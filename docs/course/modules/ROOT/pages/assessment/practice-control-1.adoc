OUTPUT_FILE = "audit.json"
def unanswered(value):
    if isinstance(value, dict):
        return not value or any(unanswered(item) for item in value.values())
    if isinstance(value, (list, tuple)):
        return not value or any(unanswered(item) for item in value)
    return value is None or (isinstance(value, str) and not value.strip())

missing = [key for key, value in responses.items() if unanswered(value)]
if missing:
    print("Rehearsal incomplete. Add your own evidence for:", ", ".join(missing))
else:
    payload = {"data_id": DATA_ID, "seed": SEED, "responses": responses}
    output = Path(OUTPUT_FILE)
    output.write_text(json.dumps(payload, indent=2, allow_nan=False) + "\n")
    print("Wrote", output.name, "— completeness only; correctness still needs review.")
