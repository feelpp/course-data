*Question for the data.* Can a tree or an average of trees improve the estimate
of same-hour PM2.5? Can either improve the decision that PM2.5 reaches
75 µg/m³?

Run the setup. Keep one row per station-hour and the five ordered predictor

[#trees-forests-setup]
[%dynamic,python,fail-on-error=]
----
from pathlib import Path

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier, RandomForestRegressor
from sklearn.linear_model import LinearRegression, LogisticRegression, Ridge
from sklearn.metrics import confusion_matrix, mean_absolute_error, root_mean_squared_error
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.tree import DecisionTreeClassifier, DecisionTreeRegressor, export_text

relative = Path("datasets/teaching/air-quality/observations.csv")
root = next((folder for folder in (Path.cwd(), *Path.cwd().parents)
             if (folder / relative).is_file()), None)
if root is None:
    raise FileNotFoundError("Open this notebook inside the course checkout or student bundle.")

air = pd.read_csv(root / relative)
air["timestamp"] = pd.to_datetime(air[["year", "month", "day", "hour"]])
predictors = ["PM10", "TEMP", "PRES", "DEWP", "WSPM"]
response = "PM2.5"
rows = air.dropna(subset=[response, *predictors]).sort_values(["timestamp", "station"])

cutoff = pd.Timestamp("2013-03-11")
train = rows.loc[rows["timestamp"] < cutoff]
later = rows.loc[rows["timestamp"] >= cutoff]
X_train, X_later = train[predictors], later[predictors]
y_train, y_later = train[response], later[response]

class_cutoff = 75.0
probability_cutoff = 0.50
tree_options = dict(max_depth=5, min_samples_leaf=10, random_state=42)
forest_options = dict(
    **tree_options, n_estimators=100, max_features="sqrt", bootstrap=True
)
----
